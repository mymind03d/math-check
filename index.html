          <button onclick="manualCheckInWithStatus('มาปกติ')" class="bg-blue-600 text-white py-2.5 rounded-xl font-medium hover:bg-blue-700 transition shadow-sm">✔ เช็คชื่อ "มาปกติ" (กดมือ)</button>

  <script>

    // 🔗 วางลิงก์สคริปต์ /exec ตัวใหม่ของคุณครูตรงนี้ครับ

    const BACKEND_URL = "https://script.googleusercontent.com/macros/echo?user_content_key=AUkAhnSnBOZSeggddn-7ncan5tezCkymBgocZz0HcseMboBFxTLZYSakA1wU-B1DOtBUh1za8CvzRCmts4uDd6VM6KSeJIY8I2lCH0_4IqkGQoI-jZ9KhR2AWJngBITxQKMCFSOKpf_BV5y2VU8EH8rMrevHLiv3EPENQNbJd5EsNUfMk7SmKRQtGT3sWlN1uc34J3S2zy70JH7WuzoYtjIB2YBdjtRXzG8h-BsRpCvJsGGKuUFspGnpRP_8g07iHkrcXOLHPbT76td5yMwPcKq97VIVwoEYIw&lib=Ml38aqEqIDCwOuHTerqOB3SkLkff0V_EX"; 


    const video = document.getElementById('video');

    const statusMsg = document.getElementById('status-message');

    let localStudents = [];

    let faceMatcher = null;


    // ฟังก์ชันส่วนกลางสำหรับเรียกคุยกับหลังบ้าน Google Sheets

    async function callBackend(payload) {

      const response = await fetch(BACKEND_URL, {

        method: "POST",

        body: JSON.stringify(payload)

      });

      return await response.json();

    }


    async function initSystem() {

      statusMsg.innerText = "⏳ กำลังโหลดโมเดลตรวจจับใบหน้าจากระบบ...";

      const MODEL_URL = 'https://cdn.jsdelivr.net/npm/@vladmandic/face-api/model/';

      await faceapi.nets.tinyFaceDetector.loadFromUri(MODEL_URL);

      await faceapi.nets.faceLandmark68Net.loadFromUri(MODEL_URL);

      await faceapi.nets.faceRecognitionNet.loadFromUri(MODEL_URL);

      statusMsg.innerText = "⏳ โมเดลพร้อมแล้ว กำลังเชื่อมต่อกล้อง...";

      startVideo();

      loadStudents();

    }


    function startVideo() {

      navigator.mediaDevices.getUserMedia({ video: { facingMode: "user", width: { ideal: 640 }, height: { ideal: 480 } } })

      .then(stream => {

        video.srcObject = stream;

        statusMsg.innerText = "🟢 ระบบสแกนใบหน้าพร้อมทำงานแล้วครับ";

        statusMsg.className = "mt-4 p-2 px-4 rounded-full text-xs font-semibold bg-emerald-50 text-emerald-600 text-center";

        video.addEventListener('play', () => { setInterval(recognizeFace, 2500); });

      })

      .catch(err => {

        statusMsg.innerText = "❌ เปิดกล้องไม่สำเร็จ กรุณากดอนุญาตให้สิทธิ์เข้าถึงกล้อง";

        statusMsg.className = "mt-4 p-2 px-4 rounded-full text-xs font-semibold bg-rose-50 text-rose-600 text-center";

      });

    }


    async function loadStudents() {

      try {

        const students = await callBackend({ action: "getStudentList" });

        localStudents = students;

        const select = document.getElementById('student-select');

        select.innerHTML = '<option value="">-- เลือกรายชื่อนักเรียน --</option>';

        const labeledDescriptors = [];


        students.forEach(s => {

          let opt = document.createElement('option');

          opt.value = s.no;

          opt.innerText = `เลขที่ ${s.no} - ${s.name} (${s.nickname})`;

          select.appendChild(opt);


          if(s.faceToken) {

            try {

              const arr = new Float32Array(JSON.parse(s.faceToken));

              labeledDescriptors.push(new faceapi.LabeledFaceDescriptors(s.no.toString(), [arr]));

            } catch(e) { console.error(e); }

          }

        });


        if(labeledDescriptors.length > 0) {

          faceMatcher = new faceapi.FaceMatcher(labeledDescriptors, 0.55);

        }

      } catch (err) { statusMsg.innerText = "❌ โหลดข้อมูลนักเรียนล้มเหลว"; }

    }


    async function recognizeFace() {

      if(!faceMatcher || isProcessing) return;

      const detections = await faceapi.detectAllFaces(video, new faceapi.TinyFaceDetectorOptions()).withFaceLandmarks().withFaceDescriptors();

      if(detections.length > 0) {

        detections.forEach(detection => {

          const bestMatch = faceMatcher.findBestMatch(detection.descriptor);

          if(bestMatch.label !== 'unknown') {

            const student = localStudents.find(s => s.no == bestMatch.label);

            if(student) triggerCheckIn(student, null);

          }

        });

      }

    }


    let isProcessing = false;

    async function triggerCheckIn(student, customStatus = null) {

      if(isProcessing) return;

      isProcessing = true;

      statusMsg.innerText = `🔔 กำลังบันทึกสถานข้อมูลของ ${student.name}...`;


      try {

        const res = await callBackend({ action: "checkAttendance", studentData: student, customStatus: customStatus });

        const resultBox = document.getElementById('result-box');

        const resultDiv = resultBox.firstElementChild;

        resultBox.classList.remove('hidden');

        document.getElementById('res-name').innerText = `🎉 ${student.name} (${student.nickname})`;

        

        resultDiv.className = "p-4 border rounded-xl text-center ";

        if(res.studentStatus === "มาปกติ") resultDiv.classList.add("bg-emerald-50", "border-emerald-200", "text-emerald-700");

        else if(res.studentStatus === "สาย") resultDiv.classList.add("bg-yellow-50", "border-yellow-200", "text-yellow-700");

        else if(res.studentStatus === "ลา" || res.studentStatus === "ป่วย") resultDiv.classList.add("bg-amber-50", "border-amber-200", "text-amber-700");

        else resultDiv.classList.add("bg-rose-50", "border-rose-200", "text-rose-700");


        document.getElementById('res-detail').innerText = `บันทึกสถานะ: ${res.studentStatus} สำเร็จ เวลา: ${res.time} น.`;

        statusMsg.innerText = `✔ บันทึกเวลาเรียนสำเร็จแล้ว`;

        setTimeout(() => { isProcessing = false; }, 4000);

      } catch (err) { statusMsg.innerText = "❌ บันทึกผลล้มเหลว"; isProcessing = false; }

    }


    async function registerFace() {

      const studentNo = document.getElementById('student-select').value;

      if(!studentNo) { alert("กรุณาเลือกรายชื่อนักเรียนก่อนครับ"); return; }

      statusMsg.innerText = "📸 กำลังประมวลผลจัดเก็บพิกัดใบหน้า...";

      const detection = await faceapi.detectSingleFace(video, new faceapi.TinyFaceDetectorOptions()).withFaceLandmarks().withFaceDescriptor();


      if(detection) {

        const tokenArray = Array.from(detection.descriptor);

        try {

          const res = await callBackend({ action: "saveFaceToken", studentNo: studentNo, faceTokenString: JSON.stringify(tokenArray) });

          alert(res.message);

          loadStudents();

        } catch(err) { alert("❌ ลงทะเบียนล้มเหลว"); }

      } else { alert("❌ ตรวจไม่พบใบหน้า กรุณายืนหน้าตรงแล้วลองใหม่ครับ"); }

    }


    function manualCheckInWithStatus(selectedStatus) {

      const studentNo = document.getElementById('student-select').value;

      if(!studentNo) { alert("กรุณาเลือกชื่อนักเรียนก่อนครับคุณครู"); return; }

      const student = localStudents.find(s => s.no == studentNo);

      if(student) triggerCheckIn(student, selectedStatus);

    }


    window.onload = initSystem;
